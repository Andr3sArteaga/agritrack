// TEMPORAL (paso 0, rama prueba/deps-reportes): prueba de humo de write-excel-file bajo Jest CJS.
import { strFromU8, unzipSync } from 'fflate';
import writeXlsxFile from 'write-excel-file/node';

describe('write-excel-file en el servidor (Jest CJS)', () => {
  it('genera un .xlsx de dos hojas con tildes, numeros y fechas', async () => {
    const buffer = await writeXlsxFile(
      [
        {
          sheet: 'Reporte',
          data: [
            [
              { value: 'Campo', fontWeight: 'bold' },
              { value: 'Valor', fontWeight: 'bold' },
            ],
            [{ value: 'Campaña' }, { value: 'Verano 2026 · Soya' }],
            [{ value: 'Rendimiento (t/ha)' }, { value: 3.4, format: '0.00' }],
            [
              { value: 'Generado' },
              { value: new Date('2026-11-02T15:30:00Z'), format: 'dd/mm/yyyy hh:mm' },
            ],
          ],
          columns: [{ width: 24 }, { width: 30 }],
        },
        {
          sheet: 'Gastos',
          data: [
            [{ value: 'Descripción' }, { value: 'Monto (Bs)' }],
            [{ value: 'Semilla de soya' }, { value: 2193.5, format: '#,##0.00' }],
          ],
        },
      ],
    ).toBuffer();

    expect(buffer.subarray(0, 2).toString('latin1')).toBe('PK');
    const archivos = unzipSync(new Uint8Array(buffer));
    const xml = Object.entries(archivos)
      .filter(([nombre]) => nombre.endsWith('.xml'))
      .map(([, contenido]) => strFromU8(contenido))
      .join('\n');
    expect(Object.keys(archivos)).toEqual(
      expect.arrayContaining(['xl/worksheets/sheet1.xml', 'xl/worksheets/sheet2.xml']),
    );
    expect(xml).toContain('Campaña');
    expect(xml).toContain('Descripción');
    expect(xml).toContain('3.4');
    expect(xml).toContain('2193.5');
  });
});
